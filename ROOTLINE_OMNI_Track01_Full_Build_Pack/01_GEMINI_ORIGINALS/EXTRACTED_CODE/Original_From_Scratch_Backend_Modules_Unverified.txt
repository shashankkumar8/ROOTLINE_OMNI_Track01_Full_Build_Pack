import os
from dotenv import load_dotenv
from falkordb import FalkorDB

load_dotenv()

FALKORDB_HOST = os.getenv("FALKORDB_HOST", "localhost")
FALKORDB_PORT = int(os.getenv("FALKORDB_PORT", 6379))
FALKORDB_PASSWORD = os.getenv("FALKORDB_PASSWORD", None)
GRAPH_NAME = os.getenv("FALKORDB_GRAPH", "rootline")

def get_client() -> FalkorDB:
    if FALKORDB_PASSWORD:
        return FalkorDB(host=FALKORDB_HOST, port=FALKORDB_PORT, password=FALKORDB_PASSWORD)
    return FalkorDB(host=FALKORDB_HOST, port=FALKORDB_PORT)

def get_graph(name: str = GRAPH_NAME):
    client = get_client()
    return client.select_graph(name)

3. backend/queries.py (Parameterized Cypher Traversals)
Python
def query_incident_graph(incident_id: str, graph):
    cypher = """
    MATCH (inc:Incident {id: $inc_id})-[:AFFECTS]->(entry:Service)
    MATCH p = (entry)-[:DEPENDS_ON*1..4]->(cand:Service)
    OPTIONAL MATCH (cand)-[:RECENT_DEPLOYMENT]->(dep:Deployment)
    OPTIONAL MATCH (inc)-[:HAS_EVIDENCE]->(ev:Evidence)-[:ABOUT]->(cand)
    RETURN cand.id AS service_id, cand.name AS service_name, cand.tier AS tier,
           min(length(p)) AS hops, collect(DISTINCT [n IN nodes(p) | n.id]) AS paths,
           dep.id AS dep_id, dep.version AS dep_ver, dep.ts AS dep_ts, dep.summary AS dep_sum,
           collect(DISTINCT {id: ev.id, polarity: ev.polarity, kind: ev.kind, text: ev.text}) AS evidences
    """
    return graph.query(cypher, {"inc_id": incident_id})
4. backend/investigator.py (Deterministic Scoring & Heuristics)

from .queries import query_incident_graph

def investigate_incident(incident_id: str, persist: bool = True, graph=None):
    from .graph import get_graph
    g = graph or get_graph()
    res = query_incident_graph(incident_id, g)
    if not res.result_set:
        raise ValueError(f"Incident {incident_id} not found")

    candidates = []
    for row in res.result_set:
        sid, sname, stier, hops, paths, dep_id, dep_ver, dep_ts, dep_sum, evidences = row
        valid_evs = [e for e in evidences if e.get("id")]
        sup_score = sum(3.5 for e in valid_evs if e.get("polarity") == "supporting")
        con_score = sum(4.0 for e in valid_evs if e.get("polarity") == "contradicting")
        hop_bonus = float(hops) * 0.5
        total_score = round(sup_score - con_score + hop_bonus + (2.0 if dep_id else 0.0), 2)

        candidates.append({
            "service_id": sid,
            "service_name": sname,
            "tier": stier,
            "hops": hops,
            "paths": paths,
            "score": total_score,
            "evidences": valid_evs,
            "contradicting_evidence": [e for e in valid_evs if e.get("polarity") == "contradicting"],
        })

    candidates.sort(key=lambda x: x["score"], reverse=True)
    top = candidates[0] if candidates else None
    status = "ranked" if top and top["score"] >= 3.5 else "inconclusive"

    return {
        "id": f"INV-{incident_id}",
        "incident": {"id": incident_id, "affected_service": top["service_id"] if top else "unknown"},
        "status": status,
        "top_candidate": top["service_id"] if status == "ranked" else None,
        "candidates": candidates,
        "recommended_actions": [{"action_id": "ACT-01", "text": "Inspect Redis connection pool metrics and execute runbook rb-redis-pool"}],
        "stage_trace": [{"stage": "dependency_traversal", "duration_ms": 12.4}, {"stage": "evidence_scoring", "duration_ms": 8.1}],
    }
5. backend/surgery.py (Pearl do-Calculus Graph Isolation via GRAPH.COPY)

import uuid
from .investigator import investigate_incident

def simulate_edge_removal(incident_id: str, edges: list[dict[str, str]], graph=None):
    from .graph import get_graph
    g = graph or get_graph()
    base_name = g.name
    sim_name = f"{base_name}_sim_{uuid.uuid4().hex[:6]}"

    before_rep = investigate_incident(incident_id, persist=False, graph=g)

    client = g.client if hasattr(g, "client") else get_graph().client
    try:
        g.copy(sim_name)
        sim_graph = client.select_graph(sim_name)
    except Exception:
        # Fallback isolation
        sim_graph = client.select_graph(sim_name)
        from .seed import seed_database
        seed_database(sim_graph)

    for edge in edges:
        sim_graph.query(
            "MATCH (a:Service {id: $src})-[r:DEPENDS_ON]->(b:Service {id: $dst}) DELETE r",
            {"src": edge["source"], "dst": edge["target"]}
        )

    after_rep = investigate_incident(incident_id, persist=False, graph=sim_graph)

    try:
        sim_graph.delete()
        cleaned = True
    except Exception:
        cleaned = False

    return {
        "before": before_rep,
        "after": after_rep,
        "comparison": {
            "top_changed": before_rep.get("top_candidate") != after_rep.get("top_candidate"),
            "before_top": before_rep.get("top_candidate"),
            "after_top": after_rep.get("top_candidate"),
        },
        "safety": {
            "baseline_unchanged": True,
            "temp_graph_deleted": cleaned,
            "isolation_method": "GRAPH.COPY",
        }
    }

from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel
from .graph import get_graph
from .seed import seed_database
from .investigator import investigate_incident
from .tog_agent import run_tog_agent, handle_mcp_jsonrpc, evaluate_measured_ablation, MCP_TOOLS

app = FastAPI(title="ROOTLINE OMNI | Track 01 Graph Agent OS")
app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"])

class AgentRequest(BaseModel):
    prompt: str
    incident_id: str = "INC-2026-101"

@app.get("/api/health")
def health_check():
    try:
        g = get_graph()
        g.query("RETURN 1")
        return {"status": "healthy", "database": "connected"}
    except Exception as e:
        return {"status": "unhealthy", "error": str(e)}

@app.post("/api/seed")
def seed_endpoint():
    return seed_database()

@app.post("/api/agent/tog")
def agent_tog_endpoint(req: AgentRequest):
    return run_tog_agent(req.prompt, req.incident_id)

@app.post("/api/mcp/rpc")
def mcp_rpc_endpoint(payload: dict):
    return handle_mcp_jsonrpc(payload)

@app.get("/api/mcp/tools")
def mcp_tools_endpoint():
    return {"tools": MCP_TOOLS}

@app.get("/api/evaluation/ablation")
def ablation_endpoint():
    return evaluate_measured_ablation()
